# source blocks require emacs-restclient
# setup: use session/post below to create a new token and paste it in the source block below
#+begin_src emacs-lisp
(setq config (list (list "addr" "http://127.0.0.1:3000")
                   (list "user" "testuser000")
                   (list "pw" "testuser-000")
                   (list "duration" 60)
                   (list "token" "paste-session-token-here")))
(defun config-get (key) (car (cdr (assoc key config))))
(setq restclient-enable-eval t)
#+end_src

** user

#+begin_src restclient

GET :addr/user/:user

*** get comments
query parameters:
- =offset=: defaults to =0=
- =limit=: defaults to =20=. one of =20=, =50=, or =100=
- =sort_by=: =newest= or =oldest=, defaults to =oldest=

#+begin_src restclient
:addr := (config-get "addr")
:id = 1

GET :addr/post/:id/comments?offset=0&limit=20&sort_by=newest
#+end_src
